    "def zip_two(a, b):\n",
    "    out = []\n",
    "    for i in range(max(len(a), len(b))):\n",
    "        if i < len(a):\n",
    "            out.append(a[i])\n",
    "        if i < len(b):\n",
    "            out.append(b[i])\n",
    "    return \"\".join(out)\n",
